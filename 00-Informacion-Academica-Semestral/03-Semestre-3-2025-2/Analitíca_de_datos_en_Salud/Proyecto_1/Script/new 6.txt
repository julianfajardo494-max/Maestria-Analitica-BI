CREATE OR REPLACE VIEW vw_tiempos_espera_oportunidad AS
SELECT 
    -- Identificadores y dimensiones básicas reales
    h.key_hecho_atencion,
    h.key_dim_ips,
    h.tiempo_espera_dias,
    h.tiempo_espera_horas,
    h.tiempo_espera_minutos,
    h.costo,
    
    -- Fechas desde la tabla transaccional de servicios relacionada
    t.fecha_solicitud,
    t.fecha_atencion,
    
    -- Cálculo directo de días de espera utilizando los campos de la tabla transaccional
    EXTRACT(EPOCH FROM (t.fecha_atencion - t.fecha_solicitud)) / 86400 AS dias_espera_calculados

FROM hecho_atencion h
LEFT JOIN trans_servicio t ON h.key_trans_servicio = t.key_trans_servicio;



CREATE OR REPLACE VIEW vw_analitica_ips_demografia AS
SELECT 
    -- Identificadores y métricas base
    h.key_hecho_atencion,
    h.costo,
    h.tiempo_espera_dias,
    
    -- Información de la IPS / Centro Hospitalario
    i.nombre AS nombre_ips,
    i.tipo_ips,
    i.municipio AS municipio_ips,
    i.departamento AS departamento_ips,
    i.nivel AS nivel_ips,
    
    -- Información Demográfica del Paciente
    d.sexo,
    d.edad,
    d.estracto,
    d.tipo_usuario,
    d.empresa,
    d.municipio AS municipio_paciente,
    d.departamento AS departamento_paciente,
    
    -- Diagnóstico asociado
    dg.diagnostico,
    dg.fecha_diagnostico

FROM hecho_atencion h
LEFT JOIN dim_ips i ON h.key_dim_ips = i.key_dim_ips
LEFT JOIN dim_demografia d ON h.key_dim_demo = d.key_dim_demo
LEFT JOIN dim_diag dg ON h.key_dim_diag = dg.key_dim_diag;
